# =========================================================
# 4) SESSION STATE
# =========================================================

DEFAULTS = {
    "page": 0,
    "date_choice": None,
    "time_choice": None,
    "game_choice": None,
    "food_choice": None,
    "confirmed": False,
    "celebrated": False,
}

for key, value in DEFAULTS.items():
    if key not in st.session_state:
        st.session_state[key] = value


def next_page():
    st.session_state.page = min(st.session_state.page + 1, 4)
    st.rerun()


def prev_page():
    st.session_state.page = max(st.session_state.page - 1, 0)
    st.rerun()


def navigation(can_continue=True, next_label="Tiếp tục →"):
    left, right = st.columns(2)

    with left:
        if st.session_state.page > 0:
            if st.button("← Quay lại", use_container_width=True):
                prev_page()

    with right:
        if st.button(
            next_label,
            type="primary",
            use_container_width=True,
            disabled=not can_continue,
        ):
            next_page()


# =========================================================
# 5) THANH TIẾN TRÌNH
# =========================================================

if st.session_state.page > 0:
    st.markdown(
        f'<div class="progress-label">Bước {st.session_state.page} / 4</div>',
        unsafe_allow_html=True,
    )
    st.progress(st.session_state.page / 4)


# =========================================================
# 6) CÁC TRANG
# =========================================================

# ---------- Trang 1: Lời mời ----------
if st.session_state.page == 0:
    st.markdown(
        """
        <div class="invite-card">
            <div class="tiny-label">💌 Một lời mời nho nhỏ</div>
            <div class="hero-title">Chào bạn thân yêu!</div>
            <div class="hero-subtitle">
                Đi chơi cùng chúng tớ nhé? 🥺💗
            </div>
            <div class="small-note">
                nếu bạn rảnh :))
            </div>
        </div>
        """,
        unsafe_allow_html=True,
    )

    st.write("")
    if st.button(
        "Okii, xem kèo gì nào ✨",
        type="primary",
        use_container_width=True,
    ):
        next_page()


# ---------- Trang 2: Chọn ngày & thời gian ----------
elif st.session_state.page == 1:
    st.markdown(
        """
        <div class="page-title">📅 Bạn rảnh hôm nào?</div>
        <div class="page-desc">
            Chọn giúp chúng tớ <b>1 ngày</b> và <b>1 khung giờ</b> nha.
        </div>
        """,
        unsafe_allow_html=True,
    )

    st.markdown('<div class="section-title">Ngày đi chơi</div>', unsafe_allow_html=True)

    date_choice = st.pills(
        "Chọn ngày",
        DATES,
        selection_mode="single",
        default=st.session_state.date_choice,
        key="date_widget",
        label_visibility="collapsed",
    )
    st.session_state.date_choice = date_choice

    st.markdown('<div class="section-title">Khung giờ bạn rảnh</div>', unsafe_allow_html=True)

    time_choice = st.pills(
        "Chọn thời gian",
        TIME_SLOTS,
        selection_mode="single",
        default=st.session_state.time_choice,
        key="time_widget",
        label_visibility="collapsed",
    )
    st.session_state.time_choice = time_choice

    can_continue = bool(date_choice and time_choice)

    if not can_continue:
        st.caption("Chọn đủ ngày và giờ rồi mình đi tiếp nha 👀")

    navigation(can_continue=can_continue)


# ---------- Trang 3: Chọn trò chơi ----------
elif st.session_state.page == 2:
    st.markdown(
        """
        <div class="page-title">🎮 Chơi gì đây ta?</div>
        <div class="page-desc">
            Chỉ được pick <b>1 trò</b> thôi nhé — lựa chọn khó khăn bắt đầu rồi đó :))
        </div>
        """,
        unsafe_allow_html=True,
    )

    game_choice = st.pills(
        "Chọn trò chơi",
        GAMES,
        selection_mode="single",
        default=st.session_state.game_choice,
        key="game_widget",
        label_visibility="collapsed",
    )
    st.session_state.game_choice = game_choice

    if not game_choice:
        st.caption("Pick một trò bạn muốn chơi nhất nha ✨")

    navigation(can_continue=bool(game_choice))


# ---------- Trang 4: Chọn đồ ăn ----------
elif st.session_state.page == 3:
    st.markdown(
        """
        <div class="page-title">🍜 Chơi xong đi ăn hong?</div>
        <div class="page-desc">
            Kèo ăn là phần rất quan trọng của chuyến đi này 😌
        </div>
        """,
        unsafe_allow_html=True,
    )

    food_choice = st.pills(
        "Chọn món ăn",
        FOODS,
        selection_mode="single",
        default=st.session_state.food_choice,
        key="food_widget",
        label_visibility="collapsed",
    )
    st.session_state.food_choice = food_choice

    if not food_choice:
        st.caption("Chọn một option nha — từ chối đi ăn cũng được 🥹")

    navigation(
        can_continue=bool(food_choice),
        next_label="Xem lại kèo →",
    )


# ---------- Trang 5: Kết ----------
elif st.session_state.page == 4:
    st.markdown(
        """
        <div class="invite-card">
            <div class="tiny-label">✨ Kèo đã lên</div>
            <div class="hero-title">Rất mong chờ chuyến đi chơi!</div>
            <div class="hero-subtitle">
                Cảm ơn bạn đã chịu khó chọn hết nè 💗
            </div>
            <div class="small-note">
                Hẹn gặp bạn ở một ngày thật vui nhé!
            </div>
        </div>
        """,
        unsafe_allow_html=True,
    )

    st.markdown(
        f"""
        <div class="summary-box">
            <div class="summary-row">📅 <b>Ngày:</b> {st.session_state.date_choice}</div>
            <div class="summary-row">⏰ <b>Thời gian:</b> {st.session_state.time_choice}</div>
            <div class="summary-row">🎮 <b>Trò chơi:</b> {st.session_state.game_choice}</div>
            <div class="summary-row">🍜 <b>Ăn uống:</b> {st.session_state.food_choice}</div>
        </div>
        """,
        unsafe_allow_html=True,
    )

    if not st.session_state.confirmed:
        col1, col2 = st.columns([1, 1.45])

        with col1:
            if st.button("← Sửa lựa chọn", use_container_width=True):
                st.session_state.page = 1
                st.rerun()

        with col2:
            if st.button(
                "Chốt kèo này! 💖",
                type="primary",
                use_container_width=True,
            ):
                st.session_state.confirmed = True
                st.rerun()

    else:
        st.success("Đã chốt kèo! Hẹn nhau hôm đó nhaaa 🫶")

        if not st.session_state.celebrated:
            st.balloons()
            st.session_state.celebrated = True

        if st.button("Chọn lại từ đầu ↻", use_container_width=True):
            for key, value in DEFAULTS.items():
                st.session_state[key] = value
            # Xóa state của widget để tránh giữ lựa chọn cũ
            for key in ["date_widget", "time_widget", "game_widget", "food_widget"]:
                st.session_state.pop(key, None)
            st.rerun()
